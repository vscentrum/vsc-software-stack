import os
from easybuild.tools.systemtools import get_os_name, get_os_version
from easybuild.tools.build_log import print_warning


def parse_hook(self, *args, **kwargs):
    # avoid OpenSSL/1.1 dependency on RHEL 9.x, replace with OpenSSL/3 (which wraps around system OpenSSL)
    os_name, os_version = get_os_name(), get_os_version()
    if os_name == 'RHEL' and os_version.startswith('9'):
        for key in ('builddependencies', 'dependencies'):
            deps = self[key]
            for idx, dep in enumerate(deps):
                if dep[0] == 'OpenSSL' and dep[1] == '1.1':
                    print_warning("Replacing OpenSSL/1.1 with OpenSSL/3 for %s %s" % (self.name, self.version))
                    dep = list(dep)
                    dep[1] = '3'
                    deps[idx] = tuple(dep)
                    
def pre_test_hook(self, *args, **kwargs):
    # unset $OMP_PROC_BIND for certain installations, to avoid that it makes the test step way too slow
    if self.name in ('PyTorch', 'CP2K', ):
        if 'OMP_PROC_BIND' in os.environ:
            del os.environ['OMP_PROC_BIND']
